% A one-line "left ... right" row. \parfillskip is zeroed locally, otherwise

  \par\vspace{1pt}%
  {\setlength{\parfillskip}{0pt}\noindent #1\par}%
}

  \fi
  \vspace{-1pt}%
  \par\noindent{\color{muted}\itshape\small #2}%
  \vspace{-3pt}%
}
